static bool AliasEquals(const char* a, const char* b) {
    while (*a != '\0' && *b != '\0') {
        const char ca = (*a >= 'A' && *a <= 'Z') ? (char)(*a - 'A' + 'a') : *a;
        const char cb = (*b >= 'A' && *b <= 'Z') ? (char)(*b - 'A' + 'a') : *b;
        if (ca != cb) return false;
        ++a;
        ++b;
    }
    return *a == '\0' && *b == '\0';
}

int DFA_W_FindClass(const char* name) {
    if (name == nullptr) return -1;

    while (*name == ' ' || *name == '\t') ++name;
    const char* end = name;
    while (*end != '\0') ++end;
    while (end > name && (end[-1] == ' ' || end[-1] == '\t')) --end;

    const int len = (int)(end - name);
    char buf[64];
    if (len <= 0 || len >= (int)sizeof(buf)) return -1;
    for (int i = 0; i < len; ++i) buf[i] = name[i];
    buf[len] = '\0';

    const int aliasCount = (int)(sizeof(kClassAliases) / sizeof(kClassAliases[0]));
    for (int i = 0; i < aliasCount; ++i) {
        if (AliasEquals(buf, kClassAliases[i].text)) return (int)kClassAliases[i].cls;
    }
    return -1;
}

bool WideToUtf8(const std::wstring& w, std::string& out)
{
    out.clear();
    if (w.empty()) return true;
    const int n = ::WideCharToMultiByte(CP_UTF8, WC_ERR_INVALID_CHARS, w.c_str(),
                                        static_cast<int>(w.size()), nullptr, 0, nullptr, nullptr);
    if (n <= 0) return false;
    out.resize(static_cast<size_t>(n));
    return ::WideCharToMultiByte(CP_UTF8, WC_ERR_INVALID_CHARS, w.c_str(),
                                 static_cast<int>(w.size()), &out[0], n, nullptr, nullptr) == n;
}

bool Utf8ToWide(const std::string& s, std::wstring& out)
{
    out.clear();
    if (s.empty()) return true;
    const int n = ::MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS, s.c_str(),
                                       static_cast<int>(s.size()), nullptr, 0);
    if (n <= 0) return false;
    out.resize(static_cast<size_t>(n));
    return ::MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS, s.c_str(),
                                 static_cast<int>(s.size()), &out[0], n) == n;
}

bool AcpToUtf8(const std::string& in, std::string& out)
{
    out.clear();
    if (in.empty()) return true;
    const int n = ::MultiByteToWideChar(CP_ACP, 0, in.c_str(), static_cast<int>(in.size()), nullptr, 0);
    if (n <= 0) return false;
    std::wstring w(static_cast<size_t>(n), L'\0');
    if (::MultiByteToWideChar(CP_ACP, 0, in.c_str(), static_cast<int>(in.size()), &w[0], n) != n) return false;
    return WideToUtf8(w, out);
}

// 严格校验 UTF-8（拒绝超长编码、代理区、越界码点）
bool IsValidUtf8(const std::string& s)
{
    const size_t n = s.size();
    size_t i = 0;
    while (i < n) {
        const unsigned char c = static_cast<unsigned char>(s[i]);
        if (c < 0x80) { ++i; continue; }
        int len = 0;
        unsigned int cp = 0;
        if ((c & 0xE0) == 0xC0)      { len = 2; cp = c & 0x1Fu; }
        else if ((c & 0xF0) == 0xE0) { len = 3; cp = c & 0x0Fu; }
        else if ((c & 0xF8) == 0xF0) { len = 4; cp = c & 0x07u; }
        else return false;
        if (i + static_cast<size_t>(len) > n) return false;
        for (int k = 1; k < len; ++k) {
            const unsigned char cc = static_cast<unsigned char>(s[i + static_cast<size_t>(k)]);
            if ((cc & 0xC0) != 0x80) return false;
            cp = (cp << 6) | (cc & 0x3Fu);
        }
        if (len == 2 && cp < 0x80u)     return false;
        if (len == 3 && cp < 0x800u)    return false;
        if (len == 4 && cp < 0x10000u)  return false;
        if (cp > 0x10FFFFu)             return false;
        if (cp >= 0xD800u && cp <= 0xDFFFu) return false;
        i += static_cast<size_t>(len);
    }
    return true;
}

// 把文件字节流解码为 UTF-8；失败时给出中文原因
bool DecodeToUtf8(const std::string& raw, std::string& out, std::string& why)
{
    const size_t n = raw.size();
    const unsigned char* p = reinterpret_cast<const unsigned char*>(raw.data());
    out.clear();
    why.clear();

    // UTF-32 BOM（先于 UTF-16 判断，因为 FF FE 00 00 也以 FF FE 开头）
    if (n >= 4 && ((p[0] == 0xFF && p[1] == 0xFE && p[2] == 0x00 && p[3] == 0x00) ||
                   (p[0] == 0x00 && p[1] == 0x00 && p[2] == 0xFE && p[3] == 0xFF))) {
        why = "暂不支持 UTF-32 编码，请另存为 UTF-8";
        return false;
    }

    if (n >= 3 && p[0] == 0xEF && p[1] == 0xBB && p[2] == 0xBF) {
        out.assign(raw, 3, std::string::npos);
        if (!IsValidUtf8(out)) { why = "UTF-8 BOM 之后的内容不是合法 UTF-8"; out.clear(); return false; }
        return true;
    }

    if (n >= 2 && p[0] == 0xFF && p[1] == 0xFE) {          // UTF-16 LE
        if ((n - 2) % 2 != 0) { why = "UTF-16 文件字节数为奇数，已损坏"; return false; }
        const size_t chars = (n - 2) / 2;
        std::wstring w(chars, L'\0');
        std::memcpy(&w[0], raw.data() + 2, chars * 2);
        if (!WideToUtf8(w, out)) { why = "UTF-16 内容包含非法代理项"; out.clear(); return false; }
        return true;
    }
    if (n >= 2 && p[0] == 0xFE && p[1] == 0xFF) {          // UTF-16 BE
        if ((n - 2) % 2 != 0) { why = "UTF-16 文件字节数为奇数，已损坏"; return false; }
        const size_t chars = (n - 2) / 2;
        std::wstring w(chars, L'\0');
        for (size_t i = 0; i < chars; ++i) {
            const unsigned char hi = p[2 + i * 2];
            const unsigned char lo = p[2 + i * 2 + 1];
            w[i] = static_cast<wchar_t>((static_cast<unsigned int>(hi) << 8) | lo);
        }
        if (!WideToUtf8(w, out)) { why = "UTF-16 内容包含非法代理项"; out.clear(); return false; }
        return true;
    }

    if (IsValidUtf8(raw)) { out = raw; return true; }       // 无 BOM 的 UTF-8 / 纯 ASCII

    // 兜底按本机代码页（中文系统即 GBK）解释
    if (AcpToUtf8(raw, out)) return true;
    why = "编码无法识别（既不是 UTF-8/UTF-16，也无法按本机 GBK 解码）";
    return false;
}

// ---------------------------------------------------------------- 文本工具

bool IsAsciiSpace(unsigned char c)
{
    return c == ' ' || c == '\t' || c == '\r' || c == '\n' || c == '\f' || c == '\v';
}

// 去掉首尾的 ASCII 空白与全角空格 U+3000
std::string Trim(const std::string& s)
{
    size_t b = 0;
    size_t e = s.size();
    for (;;) {
        if (b < e && IsAsciiSpace(static_cast<unsigned char>(s[b]))) { ++b; continue; }
        if (b + 2 < e && static_cast<unsigned char>(s[b]) == 0xE3 &&
            static_cast<unsigned char>(s[b + 1]) == 0x80 &&
            static_cast<unsigned char>(s[b + 2]) == 0x80) { b += 3; continue; }
        break;
    }
    for (;;) {
        if (e > b && IsAsciiSpace(static_cast<unsigned char>(s[e - 1]))) { --e; continue; }
        if (e >= b + 3 && static_cast<unsigned char>(s[e - 3]) == 0xE3 &&
            static_cast<unsigned char>(s[e - 2]) == 0x80 &&
            static_cast<unsigned char>(s[e - 1]) == 0x80) { e -= 3; continue; }
        break;
    }
    return s.substr(b, e - b);
}

// 把 | ， 、 ｜ 等分隔符统一成半角逗号
std::string NormalizeSeparators(const std::string& s)
{
    std::string out;
    out.reserve(s.size());
    const size_t n = s.size();
    for (size_t i = 0; i < n;) {
        const unsigned char c = static_cast<unsigned char>(s[i]);
        if (c == '|' || c == ',') { out.push_back(','); ++i; continue; }
        if (c == 0xEF && i + 2 < n) {
            const unsigned char c1 = static_cast<unsigned char>(s[i + 1]);
            const unsigned char c2 = static_cast<unsigned char>(s[i + 2]);
            if (c1 == 0xBD && c2 == 0x9C) { out.push_back(','); i += 3; continue; }  // ｜
            if (c1 == 0xBC && c2 == 0x8C) { out.push_back(','); i += 3; continue; }  // ，
        }
        if (c == 0xE3 && i + 2 < n &&
            static_cast<unsigned char>(s[i + 1]) == 0x80 &&
            static_cast<unsigned char>(s[i + 2]) == 0x81) { out.push_back(','); i += 3; continue; } // 、
        out.push_back(static_cast<char>(c));
        ++i;
    }
    return out;
}

std::vector<std::string> Split(const std::string& s, char sep)
{
    std::vector<std::string> parts;
    size_t start = 0;
    for (;;) {
        const size_t pos = s.find(sep, start);
        if (pos == std::string::npos) {
            parts.push_back(s.substr(start));
            break;
        }
        parts.push_back(s.substr(start, pos - start));
        start = pos + 1;
    }
    return parts;
}

bool IsComment(const std::string& t)
{
    if (t.empty()) return false;
    if (t[0] == '#' || t[0] == ';') return true;
    if (t.size() >= 2 && t[0] == '/' && t[1] == '/') return true;
    if (t.compare(0, 3, "\xEF\xBC\x83") == 0) return true;   // ＃
    if (t.compare(0, 3, "\xEF\xBC\x9B") == 0) return true;   // ；
    return false;
}

// 紧凑截断，避免把多字节字符切成半截
std::string Truncate(const std::string& s, size_t maxBytes)
{
    if (s.size() <= maxBytes) return s;
    size_t cut = maxBytes;
    while (cut > 0 && (static_cast<unsigned char>(s[cut]) & 0xC0) == 0x80) --cut;
    return s.substr(0, cut) + "…";
}

bool ParseInt(const std::string& t, int& out)
{
    if (t.empty()) return false;
    errno = 0;
    char* end = nullptr;
    const long v = std::strtol(t.c_str(), &end, 10);
    if (errno != 0 || end == t.c_str() || end == nullptr || *end != '\0') return false;
    if (v < -99 || v > 99) return false;
    out = static_cast<int>(v);
    return true;
}

char LowerAscii(char c)
{
    return (c >= 'A' && c <= 'Z') ? static_cast<char>(c - 'A' + 'a') : c;
}

bool EqualsAsciiCI(const std::string& a, const std::string& b)
{
    if (a.size() != b.size()) return false;
    for (size_t i = 0; i < a.size(); ++i) {
        if (LowerAscii(a[i]) != LowerAscii(b[i])) return false;
    }
    return true;
}

// ---------------------------------------------------------------- 文件读取

bool ReadWholeFile(const std::wstring& path, std::string& out, std::string& why)
{
    out.clear();
    HANDLE h = ::CreateFileW(path.c_str(), GENERIC_READ,
                             FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
                             nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr);
    if (h == INVALID_HANDLE_VALUE) { why = "无法打开文件（可能被占用）"; return false; }

    LARGE_INTEGER size;
    if (!::GetFileSizeEx(h, &size)) { ::CloseHandle(h); why = "无法读取文件大小"; return false; }
    if (size.QuadPart > kMaxFileBytes) {
        ::CloseHandle(h);
        why = "文件超过 4MB，已跳过";
        return false;
    }

    out.reserve(static_cast<size_t>(size.QuadPart) + 1);
    char buf[64 * 1024];
    for (;;) {
        DWORD got = 0;
        if (!::ReadFile(h, buf, static_cast<DWORD>(sizeof(buf)), &got, nullptr)) {
            ::CloseHandle(h);
            out.clear();
            why = "读取文件内容失败";
            return false;
        }
        if (got == 0) break;
        out.append(buf, got);
    }
    ::CloseHandle(h);
    return true;
}

struct ScannedFile {
    std::wstring fullPath;
    std::string  nameUtf8;
};

bool CollectFiles(const std::wstring& dir, std::vector<ScannedFile>& files)
{
    WIN32_FIND_DATAW fd;
    const std::wstring pattern = dir + L"\\*";
    HANDLE h = ::FindFirstFileW(pattern.c_str(), &fd);
    if (h == INVALID_HANDLE_VALUE) return false;

    do {
        if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) continue;
        if (fd.cFileName[0] == L'~') continue;                 // Office 临时文件
        const wchar_t* dot = std::wcsrchr(fd.cFileName, L'.');
        if (dot == nullptr) continue;
        if (_wcsicmp(dot, L".txt") != 0 && _wcsicmp(dot, L".csv") != 0) continue;

        ScannedFile f;
        f.fullPath = dir + L"\\" + fd.cFileName;
        if (!WideToUtf8(fd.cFileName, f.nameUtf8)) continue;    // 文件名无法转 UTF-8 就跳过
        files.push_back(f);
    } while (::FindNextFileW(h, &fd));
    ::FindClose(h);

    std::sort(files.begin(), files.end(), [](const ScannedFile& a, const ScannedFile& b) {
        return ::CompareStringOrdinal(a.fullPath.c_str(), -1, b.fullPath.c_str(), -1, TRUE) == CSTR_LESS_THAN;
    });
    return true;
}

// ---------------------------------------------------------------- 解析

bool NameExists(const std::string& name)
{
    for (int i = 0; i < DFA_W_GetWeaponCount(); ++i) {
        const WeaponInfo* w = DFA_W_GetWeapon(i);
        if (w != nullptr && w->name != nullptr && EqualsAsciiCI(name, w->name)) return true;
    }
    for (int i = 0; i < g_count; ++i) {
        if (EqualsAsciiCI(name, g_items[i].name)) return true;
    }
    return false;
}

void ProcessLine(const std::string& line, const std::string& fileName, int lineNo)
{
    const std::string t = Trim(line);
    if (t.empty() || IsComment(t)) return;

    const std::string prefix = fileName + ":" + std::to_string(lineNo) + " ";

    std::vector<std::string> fields = Split(NormalizeSeparators(t), ',');
    for (size_t i = 0; i < fields.size(); ++i) fields[i] = Trim(fields[i]);
    while (!fields.empty() && fields.back().empty()) fields.pop_back();

    if (fields.size() < 2) {
        AddError(prefix + "格式应为「枪名 | 大类 | 5 个评分」");
        return;
    }

    const std::string name = fields[0];
    if (name.empty()) {
        AddError(prefix + "枪名为空");
        return;
    }

    const int cls = DFA_W_FindClass(fields[1].c_str());
    if (cls < 0) {
        AddError(prefix + "大类无法识别：" + fields[1] + "（可用：突击步枪/冲锋枪/狙击步枪/射手步枪/霰弹枪/轻机枪/手枪/特殊武器/通用）");
        return;
    }

    std::vector<int> scores;
    for (size_t i = 2; i < fields.size(); ++i) {
        int v = 0;
        if (!ParseInt(fields[i], v)) {
            AddError(prefix + "评分必须是 -99~99 的整数，实际为：" + Truncate(fields[i], 32));
            return;
        }
        scores.push_back(v);
    }
    if (!scores.empty() && scores.size() != static_cast<size_t>(DFA_STYLE_COUNT)) {
        AddError(prefix + "评分需写 5 个（跑刀,鼠鼠,正常玩,微交战,全装猛攻），实际 " +
                 std::to_string(scores.size()) + " 个");
        return;
    }

    if (NameExists(name)) {
        AddError(prefix + "枪名与已有数据重复，已跳过：" + name);
        return;
    }
    if (g_count >= kMaxWeapons) {
        AddError(prefix + "自定义枪械已达上限 " + std::to_string(kMaxWeapons) + " 条，本条被忽略");
        return;
    }

    CustomWeapon& item = g_items[g_count];
    item.name   = name;
    item.source = fileName;
    item.info.name    = item.name.c_str();
    item.info.subType = item.source.c_str();
    item.info.cls     = static_cast<WeaponClass>(cls);

    const int* def = DFA_W_GetStyleScores(static_cast<WeaponClass>(cls));
    for (int s = 0; s < DFA_STYLE_COUNT; ++s) {
        item.scores[s] = (def != nullptr) ? def[s] : 0;
    }
    if (!scores.empty()) {
        for (int s = 0; s < DFA_STYLE_COUNT; ++s) item.scores[s] = scores[static_cast<size_t>(s)];
    }
    ++g_count;
}

void ProcessFile(const ScannedFile& file)
{
    std::string raw;
    std::string why;
    if (!ReadWholeFile(file.fullPath, raw, why)) {
        AddError(file.nameUtf8 + "：" + why);
        return;
    }

    std::string text;
    if (!DecodeToUtf8(raw, text, why)) {
        AddError(file.nameUtf8 + "：" + why);
        return;
    }

    size_t pos = 0;
    int lineNo = 0;
    for (;;) {
        const size_t nl = text.find('\n', pos);
        const std::string line = (nl == std::string::npos) ? text.substr(pos)
                                                           : text.substr(pos, nl - pos);
        ++lineNo;
        ProcessLine(line, file.nameUtf8, lineNo);
        if (nl == std::string::npos) break;
        pos = nl + 1;
    }
}

// ---------------------------------------------------------------- 默认目录

std::wstring ExeDirectory()
{
    std::wstring path(MAX_PATH, L'\0');
    for (;;) {
        const DWORD n = ::GetModuleFileNameW(nullptr, &path[0], static_cast<DWORD>(path.size()));
        if (n == 0) return std::wstring();
        if (n < path.size()) { path.resize(n); break; }
        path.resize(path.size() * 2);
    }
    const size_t slash = path.find_last_of(L"\\/");
    if (slash != std::wstring::npos) path.resize(slash);
    return path;
}

extern "C" {

int DFA_CW_Load(const char* folderUtf8)
{
    try {
        for (int i = 0; i < g_count; ++i) g_items[i] = CustomWeapon();
        g_count      = 0;
        g_fileCount  = 0;
        g_errorCount = 0;
        g_errorTotal = 0;
        g_folder.clear();

        if (folderUtf8 == nullptr || *folderUtf8 == '\0') return -1;

        std::wstring dir;
        if (!Utf8ToWide(folderUtf8, dir)) return -1;
        while (!dir.empty() && (dir.back() == L'\\' || dir.back() == L'/')) dir.pop_back();
        if (dir.empty()) return -1;

        const DWORD attr = ::GetFileAttributesW(dir.c_str());
        if (attr == INVALID_FILE_ATTRIBUTES || (attr & FILE_ATTRIBUTE_DIRECTORY) == 0) return -1;

        if (!WideToUtf8(dir, g_folder)) g_folder = folderUtf8;

        std::vector<ScannedFile> files;
        if (!CollectFiles(dir, files)) return -1;
        g_fileCount = static_cast<int>(files.size());
        for (size_t i = 0; i < files.size(); ++i) ProcessFile(files[i]);
        return g_count;
    } catch (...) {
        AddError("载入过程中出现内存异常，已尽可能保留已解析的数据");
        return g_count;
    }
}

#include "dfa_economy.h"

// ---------- 资金分档阈值（单位：万） ----------
// 规则：< T0 → 紧张；T0~T1 → 小康；T1~T2 → 宽裕；>= T2 → 雄厚
static const long long kWealthTiers[3] = { 100, 500, 1500 };
static const int kTierCount = (int)(sizeof(kWealthTiers) / sizeof(kWealthTiers[0])) + 1;

static const char* kWealthTierNames[] = {
    "紧张（<100万）", "小康（100~500万）", "宽裕（500~1500万）", "雄厚（≥1500万）"
};
static const char* kWealthTierNotes[] = {
    "资金紧张，应以低成本玩法保本攒钱，切忌高投入",
    "可稳定支撑机密起装（约 40~60w）成本，属于稳健积累期",
    "可承担中高端玩法试错，具备接战型玩法的资金基础",
    "资金雄厚，具备猛攻玩法的容错空间（社区观点：资产 1000 万+ 再考虑高投入）"
};

// ---------- 资金分档对五种玩法的加分（列顺序 = DfaStyle） ----------
static const int kWealthScore[][DFA_STYLE_COUNT] = {
    /* 紧张   */ {  4,  3, -2, -3, -4 },
    /* 小康   */ {  0,  1,  2,  0, -1 },
    /* 宽裕   */ { -2, -1,  2,  2,  1 },
    /* 雄厚   */ { -4, -3,  0,  2,  3 },
};

// ---------- 固定队友评分（0 = 无固排，1 = 有固排） ----------
static const int kTeamScore[2][DFA_STYLE_COUNT] = {
    /* 无固排 */ {  2,  2,  1,  0, -2 },
    /* 有固排 */ { -2, -2,  1,  2,  3 },
};

int DFA_E_WealthTier(long long wan) {
    if (wan < kWealthTiers[0]) return 0;
    if (wan < kWealthTiers[1]) return 1;
    if (wan < kWealthTiers[2]) return 2;
    return 3;
}

int DFA_E_GetTierCount() { return kTierCount; }

long long DFA_E_GetTier(int i) {
    if (i < 0 || i >= kTierCount - 1) return 0;
    return kWealthTiers[i];
}

const char* DFA_E_GetTierName(int tier) {
    if (tier < 0 || tier >= kTierCount) return "未知";
    return kWealthTierNames[tier];
}

const char* DFA_E_GetTierNote(int tier) {
    if (tier < 0 || tier >= kTierCount) return "";
    return kWealthTierNotes[tier];
}

const int* DFA_E_GetWealthStyleScores(int tier) {
    static const int kZero[DFA_STYLE_COUNT] = { 0, 0, 0, 0, 0 };
    if (tier < 0 || tier >= kTierCount) return kZero;
    return kWealthScore[tier];
}

const int* DFA_E_GetTeamStyleScores(bool hasTeam) {
    return kTeamScore[hasTeam ? 1 : 0];
}

// ============================================================================
//  dfa_engine.cpp —— 推荐引擎模块实现
//  评分模型：总分 = 枪械类别分 + 干员兵种分 + 地图分 + 难度分 + 资金分档分 + 队友分
//  强制规则：绝密 / 超临界 + 资金 < 100 万 → 强制跑刀（防破产兜底）
//  扩展：新增规则 → 在本文件底部继续追加 ForceRule 条件即可。
// ============================================================================
#include "dfa_engine.h"
#include "dfa_weapons.h"
#include "dfa_custom_weapons.h"
#include "dfa_maps.h"
#include "dfa_economy.h"
#include "dfa_operators.h"

void DFA_E_ComputeScores(const DfaProfile* p, int out[DFA_STYLE_COUNT]) {
    if (!p || !out) return;

    // 枪械类别：weaponIdx 为「内置枪械 + 自定义枪械」的统一菜单下标。
    // 内置在前（0 .. DFA_W_GetWeaponCount()-1），自定义从其后开始。
    const int builtinCount = DFA_W_GetWeaponCount();
    const int* ws = nullptr;
    if (p->weaponIdx >= 0 && p->weaponIdx < builtinCount) {
        const WeaponInfo* w = DFA_W_GetWeapon(p->weaponIdx);
        ws = DFA_W_GetStyleScores(w ? w->cls : WC_GENERIC);
    } else {
        ws = DFA_CW_GetStyleScores(p->weaponIdx - builtinCount);
        if (!ws) ws = DFA_W_GetStyleScores(WC_GENERIC);   // 越界按通用处理
    }

    // 干员兵种（越界按 0 号处理）
    const OperatorInfo* op = DFA_O_GetOperator(p->opIdx < 0 || p->opIdx >= DFA_O_GetOperatorCount() ? 0 : p->opIdx);
    const int* os = DFA_O_GetRoleScores(op ? op->role : OP_ROLE_SUPPORT);

    // 地图 / 难度（越界按 0 号处理）
    const int* ms = DFA_M_GetMapStyleScores(p->mapIdx < 0 || p->mapIdx >= DFA_MAP_COUNT ? 0 : p->mapIdx);
    const int* ds = DFA_M_GetDiffStyleScores(p->diffIdx < 0 || p->diffIdx >= DFA_DIFF_COUNT ? 0 : p->diffIdx);

    // 资金分档 + 队友
    int tier = DFA_E_WealthTier(p->wealthWan);
    const int* hs = DFA_E_GetWealthStyleScores(tier);
    const int* ts = DFA_E_GetTeamStyleScores(p->team);

    for (int s = 0; s < DFA_STYLE_COUNT; ++s) {
        out[s] = ws[s] + os[s] + ms[s] + ds[s] + hs[s] + ts[s];
    }
}

int DFA_E_PickBest(const int scores[DFA_STYLE_COUNT]) {
    if (!scores) return 0;
    int best = 0;
    for (int s = 1; s < DFA_STYLE_COUNT; ++s) {
        if (scores[s] > scores[best]) best = s;   // 同分保留编号小（风险低）
    }
    return best;
}

bool DFA_E_ForceRule(const DfaProfile* p, int& best) {
    if (!p) return false;

    // 绝密 / 超临界 + 资金不足 100 万 → 强制跑刀（防破产兜底）
    const bool highRisk = (p->diffIdx == DFA_DIFF_TOP_SECRET ||
        p->diffIdx == DFA_DIFF_SUPERCRITICAL);
    if (highRisk && p->wealthWan < 100) {
        best = DFA_STYLE_KNIFE_RUN;   // 强制跑刀
        return true;
    }
    return false;
}

// 命令行/环境变量里的路径按本机 ANSI（中文系统即 GBK）解释，再转成 UTF-8 交给模块
static std::string LocalToUtf8(const std::string& s) {
    if (s.empty()) return s;
    const int n = MultiByteToWideChar(CP_ACP, 0, s.c_str(), (int)s.size(), nullptr, 0);
    if (n <= 0) return s;
    std::wstring w((size_t)n, L'\0');
    if (MultiByteToWideChar(CP_ACP, 0, s.c_str(), (int)s.size(), &w[0], n) != n) return s;
    const int m = WideCharToMultiByte(CP_UTF8, 0, w.c_str(), (int)w.size(), nullptr, 0, nullptr, nullptr);
    if (m <= 0) return s;
    std::string u((size_t)m, '\0');
    if (WideCharToMultiByte(CP_UTF8, 0, w.c_str(), (int)w.size(), &u[0], m, nullptr, nullptr) != m) return s;
    return u;
}

// 数据目录优先级：--weapon-dir 参数 > 环境变量 DFA_WEAPON_DIR > exe 同级「自定义枪械」
static std::string ResolveWeaponFolder(int argc, char** argv) {
    for (int i = 1; i < argc; ++i) {
        const std::string arg = argv[i];
        if (arg == "--weapon-dir" && i + 1 < argc) return LocalToUtf8(argv[++i]);
        const std::string key = "--weapon-dir=";
        if (arg.compare(0, key.size(), key) == 0) return LocalToUtf8(arg.substr(key.size()));
    }
    const char* env = std::getenv("DFA_WEAPON_DIR");
    if (env != nullptr && *env != '\0') return LocalToUtf8(env);
    return DFA_CW_GetDefaultFolder();
}

static void PrintCustomStatus(const std::string& folder, int loaded) {
    if (loaded < 0) {
        std::cout << "  [自定义枪械] 未找到数据文件夹：" << folder << "\n";
        std::cout << "    可新建该文件夹放入 .txt 数据，或用 --weapon-dir <目录> 指定。\n\n";
        return;
    }
    if (loaded == 0) {
        std::cout << "  [自定义枪械] 已扫描 " << DFA_CW_GetFileCount() << " 个文件，未解析到有效数据。\n";
    } else {
        std::cout << "  [自定义枪械] 已从 " << DFA_CW_GetFileCount() << " 个文件载入 "
                  << loaded << " 把私人枪械（" << folder << "）。\n";
    }
    const int errCount = DFA_CW_GetErrorCount();
    if (errCount > 0) {
        const int show = errCount > 5 ? 5 : errCount;
        std::cout << "    发现 " << errCount << " 条数据问题，前 " << show << " 条：\n";
        for (int i = 0; i < show; ++i) {
            std::cout << "      - " << DFA_CW_GetError(i) << "\n";
        }
    }
    std::cout << "\n";
}

static int ReadInt(const char* prompt, int lo, int hi) {
    int v = 0;
    while (true) {
        std::cout << prompt;
        if (std::cin >> v) {
            if (v >= lo && v <= hi) return v;
            std::cout << "  >> 输入无效：请输入 " << lo << " ~ " << hi << " 之间的数字。\n";
        } else {
            std::cin.clear();
            std::cin.ignore(10000, '\n');
            std::cout << "  >> 输入无效：请输入数字。\n";
        }
    }
}

            case WC_LMG:     clsNote = "持续火力压制，适合阵地推进，机动性一般"; break;
            case WC_PISTOL:  clsNote = "说明预算有限或主动降风险，低成本玩法更匹配"; break;
            case WC_SPECIAL: clsNote = "特殊武器（复合弓）安静隐蔽，适合摸人与低成本探索玩法"; break;
            default:         clsNote = "无特别倾向，玩法不受枪械类型限制"; break;
        }
    }

    std::cout << "------------------------ 判断依据 ------------------------\n";
    std::cout << "  1. 枪械  ：「" << (w ? w->name : "?") << "」属于"
              << (w ? DFA_W_GetClassName(w->cls) : "?") << "（二级分类："
              << (w ? w->subType : "?") << "），"
              << (clsNote ? clsNote : "未知") << "\n";
    if (op) {
        std::cout << "  2. 干员  ：「" << op->code << "」（" << op->name << " · "
                  << DFA_O_GetRoleName(op->role) << "兵）\n";
        std::cout << "      核心技能：" << op->skills << "\n";
        std::cout << "      适配提示：" << op->advice << "\n";
    }
    std::cout << "  3. 地图  ：" << (m ? m->name : "?") << " —— " << (m ? m->note : "?") << "\n";
    std::cout << "  4. 难度  ：" << (d ? d->name : "?") << " —— " << (d ? d->note : "?") << "\n";
    std::cout << "  5. 资金  ：" << p.wealthWan << " 万（" << DFA_E_GetTierName(tier)
              << "）—— " << DFA_E_GetTierNote(tier) << "\n";
    std::cout << "  6. 队友  ：" << (p.team ? "有固定队友，可执行协同战术，高对抗玩法（微交战/全装猛攻）收益更高"
                                           : "暂无固定队友，独狼更适合稳健或低成本玩法，接战更依赖个人判断") << "\n";
    std::cout << "---------------------------------------------------------\n\n";
}

static void PrintGuide(const DfaProfile& p, int best) {
    const StyleGuide* g = DFA_S_GetGuide(best);
    if (!g) return;
    std::cout << "------------------------ 玩法档案 ------------------------\n";
    std::cout << "  【装备建议】\n    " << g->loadout << "\n\n";
    std::cout << "  【战术要点】\n    " << g->tactics << "\n\n";

    // 核心物资区随用户选择的地图联动
    const char* zones = DFA_M_GetLootZones(p.mapIdx);
    if (zones && zones[0]) {
        std::cout << "  【核心物资区】\n    " << zones << "\n\n";
    }
    const char* special = DFA_M_GetSpecialNote(p.mapIdx, p.diffIdx);
    if (special && special[0]) {
        std::cout << "  【超临界特别提示】\n    " << special << "\n\n";
    }

    std::cout << "  【经济规划】\n    " << g->economy << "\n\n";
    std::cout << "  【风险提醒】\n    " << g->risk << "\n";
    std::cout << "---------------------------------------------------------\n\n";
}

static void PrintWarnings(const DfaProfile& p, int best) {
    const OperatorInfo* op = DFA_O_GetOperator(p.opIdx);
    int tier = DFA_E_WealthTier(p.wealthWan);
    bool any = false;
    std::cout << "------------------------ 风险与提示 ----------------------\n";
    if ((p.diffIdx == DFA_DIFF_TOP_SECRET || p.diffIdx == DFA_DIFF_SUPERCRITICAL)
        && p.wealthWan < 500) {
        std::cout << "  [高危] 资金不足 500 万打绝密/超临界风险极高，建议先降一档练手攒钱。\n";
        any = true;
    }
    if (best == DFA_STYLE_FULL_PUSH && p.wealthWan < 500) {
        std::cout << "  [预警] 资金偏紧（<500万），全装一局成本 100w+，建议先降档到「微交战」。\n";
        any = true;
    }
    if (best == DFA_STYLE_FULL_PUSH && !p.team) {
        std::cout << "  [预警] 无固定队友单排全装猛攻风险高，建议先组到固定队再猛攻。\n";
        any = true;
    }
    if ((best == DFA_STYLE_KNIFE_RUN || best == DFA_STYLE_RAT) && tier == 3) {
        std::cout << "  [提示] 资金已雄厚仍跑刀/当鼠鼠效率偏低，可考虑「正常玩/微交战」提升收益。\n";
        any = true;
    }
    if (p.mapIdx == DFA_MAP_SPACE_BASE && best == DFA_STYLE_KNIFE_RUN) {
        std::cout << "  [提示] 航天基地是高强度图，跑刀收益差且易被清图队针对，慎选。\n";
        any = true;
    }
    if (p.mapIdx == DFA_MAP_TIDE_PRISON && best == DFA_STYLE_KNIFE_RUN) {
        std::cout << "  [提示] 潮汐监狱仅开放绝密，跑刀/低装玩法不适用，建议至少「正常玩」。\n";
        any = true;
    }
    if (p.diffIdx == DFA_DIFF_NIGHT) {
        std::cout << "  [提示] 永夜（黑夜）玩法视野受限，建议携带夜视仪并优先听声辨位。\n";
        any = true;
    }
    if (op && op->role == OP_ROLE_SUPPORT && !p.team) {
        std::cout << "  [提示] 支援干员独狼时团队价值有限，建议组固定队或改选进攻/侦察干员。\n";
        any = true;
    }
    if (p.diffIdx == DFA_DIFF_NIGHT && op && op->role == OP_ROLE_RECON) {
        std::cout << "  [提示] 侦察干员在永夜（黑夜）环境信息优势明显，收益更高。\n";
        any = true;
    }
    if (!any) {
        std::cout << "  当前条件下无明显风险预警，按推荐玩法执行即可。\n";
    }
    std::cout << "---------------------------------------------------------\n\n";
}

static void PrintFooter() {
    std::cout << "=============================================================\n";
    std::cout << "  说明：本程序依据游戏内设定与玩家社区共识编写，非官方数据。\n";
    std::cout << "  各模块数据均为独立工程中的「配置区」常量，修改后重新编译即可。\n";
    std::cout << "  工程结构：DFA_Weapons / DFA_CustomWeapons（私人枪械）/ DFA_Operators /\n";
    std::cout << "            DFA_Maps / DFA_Economy / DFA_Styles / DFA_Engine / DFA_Main\n";
    std::cout << "  私人枪械：把 .txt 放进 exe 同级「自定义枪械」文件夹，或 --weapon-dir <目录>。\n";
    std::cout << "=============================================================\n\n";
}

// ============================ 主流程 ============================

int main(int argc, char** argv) {
    InitConsole();
    PrintBanner();

    const std::string weaponFolder = ResolveWeaponFolder(argc, argv);
    PrintCustomStatus(weaponFolder, DFA_CW_Load(weaponFolder.c_str()));

    bool again = true;
    while (again) {
        std::cout << "请依次输入以下 6 项信息（数字菜单或 y/n）：\n\n";

        // 【1/6】枪械（内置按大类分组，自定义枪械作为最后一组，编号连续）
        int wCount = TotalWeaponCount();
        std::cout << "【1/6】最擅长 / 精通的枪械（输入序号）：\n";
        {
            int lastCls = -1;
            for (int i = 0; i < DFA_W_GetWeaponCount(); ++i) {
                const WeaponInfo* w = DFA_W_GetWeapon(i);
                if (w->cls != lastCls) {
                    lastCls = w->cls;
                    std::cout << "  【" << DFA_W_GetClassName(w->cls) << "】\n";
                }
                std::cout << "    " << (i + 1) << ". " << w->name
                          << "（" << w->subType << "）\n";
            }
            if (DFA_CW_GetCount() > 0) {
                std::cout << "  【自定义枪械（" << DFA_CW_GetFolder() << "）】\n";
                for (int i = 0; i < DFA_CW_GetCount(); ++i) {
                    const WeaponInfo* w = DFA_CW_GetWeapon(i);
                    std::cout << "    " << (DFA_W_GetWeaponCount() + i + 1) << ". " << w->name
                              << "（" << w->subType << " · " << DFA_W_GetClassName(w->cls) << "）\n";
                }
            }
        }
        std::cout << "  请输入 1 ~ " << wCount << " 之间的序号：";
        int weapon = ReadInt("", 1, wCount) - 1;

        // 【2/6】干员（按兵种分组展示，编号连续）
        int oCount = DFA_O_GetOperatorCount();
        std::cout << "\n【2/6】主玩 / 精通的干员（输入序号）：\n";
        {
            int lastRole = -1;
            for (int i = 0; i < oCount; ++i) {
                const OperatorInfo* op = DFA_O_GetOperator(i);
                if (op->role != lastRole) {
                    lastRole = op->role;
                    std::cout << "  【" << DFA_O_GetRoleName(op->role) << "兵】\n";
                }
                std::cout << "    " << (i + 1) << ". " << op->code
                          << "（" << op->name << "）\n";
            }
        }
        std::cout << "  请输入 1 ~ " << oCount << " 之间的序号：";
        int opIdx = ReadInt("", 1, oCount) - 1;

        // 【3/6】地图
        int mCount = DFA_M_GetMapCount();
        std::cout << "\n【3/6】常去的地图（输入序号）：\n";
        for (int i = 0; i < mCount; ++i) {
            const MapInfo* m = DFA_M_GetMap(i);
            std::cout << "    " << (i + 1) << ". " << m->name << "\n";
        }
        std::cout << "  请输入 1 ~ " << mCount << " 之间的序号：";
        int map = ReadInt("", 1, mCount) - 1;

        // 【4/6】难度（只列出所选地图实际开放的难度）
        int diff = 0;
        {
            const MapInfo* mInfo = DFA_M_GetMap(map);
            int dCount = DFA_M_GetAllowedDiffCount(map);
            std::cout << "\n【4/6】常玩的难度（" << (mInfo ? mInfo->name : "?")
                      << " 开放难度，输入序号）：\n";
            for (int i = 0; i < dCount; ++i) {
                int did = DFA_M_GetAllowedDiff(map, i);
                const DiffInfo* d = DFA_M_GetDiff(did);
                std::cout << "    " << (i + 1) << ". " << (d ? d->name : "?") << "\n";
            }
            std::cout << "  请输入 1 ~ " << dCount << " 之间的序号：";
            int diffSlot = ReadInt("", 1, dCount) - 1;
            diff = DFA_M_GetAllowedDiff(map, diffSlot);   // 全局难度编号
        }

        // 【5/6】资金
        std::cout << "\n【5/6】当前哈夫币存量（单位：万，例如 500 表示 500 万哈夫币）：";
        long long wealth = ReadLL("  ");

        // 【6/6】队友
        std::cout << "\n【6/6】是否有固定组队队友？(y/n)：";
        bool team = ReadYesNo("  ");

        std::cout << "\n\n";

        DfaProfile p = { weapon, opIdx, map, diff, wealth, team };

        // ---- 评估 ----
        int scores[DFA_STYLE_COUNT];
        DFA_E_ComputeScores(&p, scores);
        int best = DFA_E_PickBest(scores);
        bool forced = DFA_E_ForceRule(&p, best);

        // ---- 输出 ----
        PrintProfile(p);
        PrintScores(scores);
        PrintConclusion(best, forced);
        PrintJudgement(p);
        PrintGuide(p, best);
        PrintWarnings(p, best);
        PrintFooter();

        std::cout << "是否重新评估？(y/n)：";
        again = ReadYesNo("  ");
        std::cout << "\n\n";
    }

    std::cout << "感谢使用，祝干员局局大红！按回车键退出...";
    std::cin.ignore(10000, '\n');
    std::cin.get();
    return 0;
}
